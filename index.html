void trackAndSuppress(int leftIR, int centerIR, int rightIR) {
  if (centerIR < 320 || leftIR < 320 || rightIR < 320) {
    aimDualAxisServo(leftIR, centerIR, rightIR);
    digitalWrite(PUMP_RELAY_PIN, HIGH);
    publishEspTelemetry("FLAME_LOCKED_SUPPRESSING");
  } else {
    digitalWrite(PUMP_RELAY_PIN, LOW);
  }
}
